;obilazak po vrstama od gore do dole
.386
.model flat,c
.data
	;smestanje vrsta->vrsta
	mat byte 1,2,3,5
		byte 8,11,4,6
		byte 7,9,10,12
	n dword 3
	m dword 4
.stack 4096
.code

main proc

	mov eax,0
	mov ebx,0
	mov al,byte ptr m
	mov bl,byte ptr n
	dec bl
	mul bl
	mov dx,ax
	mov ebx,0
	wile:
		cmp bx,dx
		ja van
		mov ecx,m
		mov esi,0
		unutrasnja:
			mov eax,0
			mov al,mat[ebx][esi]
			inc esi
		loop unutrasnja
		add ebx,m
	jmp wile
	van:
	ret
main endp
end main
