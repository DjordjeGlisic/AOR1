;matrica je smestena kolona->kolona
;obilazi se po sporednoj dijagonali od dole na gore
.386
.model flat,c
.data
	;smestanje kolona->kolona
	mat byte 1,8,7,1
		byte 2,11,9,2
		byte 3,4,10,3
		byte 5,6,12,4
	n dword 4
.stack 4096
.code

main proc
	mov edx,0
	mov eax,0
	mov ebx,0
	mov esi,0
	mov bl,byte ptr n
	mov al,bl
	dec al
	mul bl
	mov dx,ax
	mov eax,0
	mov ebx,0
	mov ecx,0
	wile:
		cmp ebx,edx
		ja van
		mov esi,n
		dec esi
		sub esi,ecx
		mov al,mat[ebx][esi]
		add ebx,n
		inc ecx
	jmp wile
	van:

	ret
main endp
end main
