main proc
	mov edx,0
	mov eax,0
	mov ebx,0
	mov esi,0
	mov bl,byte ptr n
	mov al,bl
	dec al
	mul bl
	mov dx,ax
	mov eax,0
	mov ebx,0
	mov edi,edx
	wile:
	cmp ebx,edi
	ja van
	mov esi,0
	mov ecx,n
	petlja:
	mov eax,0
	mov al,byte ptr n
	mov edx,esi
	mul dl
	cmp bx,ax
	jne ink
	mov al,mat[ebx][esi]
	ink:
	inc esi
	loop petlja
	add ebx,n
	jmp wile
	van:

	ret
main endp
end main
