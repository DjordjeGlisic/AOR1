	niz db 1,2,3,4,5
	n db 5
	sum dw ?

.stack 4096
.code
	sumiraj proc
		push ebp
		mov ebp,esp
		push eax
		push ebx
		push ecx
		push edx
		push esi
		pushfd
		mov edx,0;sum=0
		mov esi,[ebp+8];adresa gde se nalazi n
		mov ecx,[esi];vrednost na adresi gde se nalazi n
		mov ebx,0
		mov bl,cl;n=5
		mov eax,0;int i=0
		mov esi,[ebp+12];niz
		petlja:
		add dl,[esi+eax]
		adc dh,0
		inc eax;i++
		loop petlja
		mov eax,[ebp+16]
		mov [eax],edx
		popfd
		pop esi
		pop edx
		pop ecx
		pop ebx
		pop eax
		pop ebp
		

	
		ret 8
	sumiraj endp

	main proc
		lea eax,sum
		push eax
		push offset niz
		lea eax,n
		push eax
		call sumiraj
		pop eax
		mov eax,[eax]
		
		ret
	main endp
	end main
