.386
.model flat,c
.data
	;smestanje kolona->kolona
	mat byte 1,2,3,5
		byte 8,11,4,6
		byte 7,9,10,12
	n dword 3
	m dword 4
.stack 4096
.code

main proc
	mov edx,0
	mov eax,0
	mov ebx,0
	mov al,byte ptr n
	dec al
	mov bl,byte ptr m
	mul bl
	mov dx,ax
	mov eax,0
	mov ebx,0

	;for(esi=0;esi<m;esi++)
	mov esi,0
	mov ecx,m
	petlja:
		mov ebx,0
		wile:
			cmp ebx,edx
			ja van
			mov al,mat[ebx][esi]
			add ebx,m
			jmp wile
		van:
		inc esi
	loop petlja

	
	ret
main endp
end main
